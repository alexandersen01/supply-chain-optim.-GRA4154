\begin{itemize}
    \item Variables = what the model is allowed to choose 
    \item Parameters = fixed numbers the model is given
\end{itemize}

\subsection{Complete Model}
\subsubsection{Sets and Indices}
\begin{align}
    \mathcal{P} &= \{N, S\} \quad\text{: plants}\\
    \mathcal{W} &= \{N, C, S\}\quad\text{: warehouses ($N, S $ co-located with plants)}\\
    \mathcal{D} &= \{1, \dots, 5\}\quad\text{: sales districts}\\
    \mathcal{T} &= \{1, \dots, 4\}\quad\text{: quarters}\\
    \mathcal{B} &\quad\text{: boats}\\
    \mathcal{B}^H &= \{\text{Anna}, \text{Emma}\} \subseteq \mathcal{B} \quad \text{(boats eligible for Q2+Q3-only rental)}\\
    \mathcal{R} &= \{1, \dots, 9\}\quad\text{: routes}\\
    \mathcal{L}_r&\quad\text{: legs of route $r $, indexed by $l $}\\
    \mathcal{R}^C&\subseteq \mathcal{R} \quad\text{: routes that visit the central warehouse $\{1, 2, 5, 6, 9\}$}\\
    \text{orig}(r, l)&\in \mathcal{W}\quad\text{: origin warehouse}\\
    \text{dest}(r, l)&\in \mathcal{W}\quad\text{: destination warehouse}\\
    \mathcal{V}(r, l)&\subseteq \mathcal{D}\quad\text{: districts visited on leg $l $}\\
    D_r&\quad\text{: total route duration (days)}
\end{align}
\subsubsection{Parameters}
\begin{align}
    \text{days}_t&\quad\text{: days in quarter $t $ (90, 91, 92, 92)}\\
    d_{j, t}&\quad\text{: demand (tons), in district $j $ during quarter $t $}\\
    \chi_p&\quad\text{: production capacity per shift per quarter at plant $p $}\\
    c_p^{\text{raw}}&\quad\text{: raw material cost per ton at plant $p $}\\
    \bar{s}_w&\quad\text{: storage capacity at warehouse $w $}\\
    \bar{h}_w&\quad\text{: handling capacity at warehouse $w $ per quarter}\\
    \bar{L}_b&\quad\text{: max load of boat $b $ (tons)}\\
    c^{\text{year}}_b, c^{\text{half}}_b&\quad\text{: annual / Q2+Q3 boat rental cost}\\
    c^{\text{shift}}&=4\text{M}\quad\text{: base cost per shift}\\
    c^{\text{ot4}} &= 1.5\text{M}\quad\text{: overtime cost for $4^{\text{th}} $ shift}\\
    c^{\text{ot5}} &= 3\text{M}\quad\text{: overtime cost for $5^{\text{th}} $ shift}\\
    c^{\text{store}}&= 50\quad\text{: storage cost per ton}\\
    c^{\text{pen}}&= 12000\quad\text{: penalty per ton for not delivering}\\
    c^{\text{rent}}&= 7\text{M}\quad\text{: rental of central warehouse}
\end{align}

\subsubsection{Decision Variables}
\begin{align}
    \delta^4_{p, t} &= 
    \begin{cases}
        1 & \quad\text{if shift 4 is run at plant $p$ in quarter $t$} \\
        0 & \quad\text{otherwise}
    \end{cases}\\
    \delta^5_{p, t} &= 
    \begin{cases}
        1 & \quad\text{if shift 5 is run at plant $p$ in quarter $t$} \\
        0 & \quad\text{otherwise}
    \end{cases}\\
    n_{p, t}&= 3 + \delta^4_{p, t} + \delta^5_{p, t}\quad\text{number of shifts at plant $p $ in quarter $t $}\\
    x_{p, t}&\ge 0 \quad\text{: tons produced at plant $p $ in quarter $t $}\\
    y &= 
    \begin{cases}
        1 & \quad\text{if central warehouse is rented} \\
        0 & \quad\text{otherwise}
    \end{cases}\\
    z_b &= 
    \begin{cases}
        1 & \quad\text{if boat rented for full year} \\
        0 & \quad\text{otherwise}
    \end{cases}\\
    z_b^H &= 
    \begin{cases}
        1 & \quad\text{if boat rented for Q2+Q3 (Anna, Emma)} \\
        0 & \quad\text{otherwise}
    \end{cases}\\
    \alpha_{b, r, t} &= 
    \begin{cases}
        1 & \quad\text{if boat $b $ assigned to route $r $ in quarter $ t $} \\
        0 & \quad\text{otherwise}
    \end{cases}\\
    f_{b, r, l, j, t}&\ge 0 \quad\text{: tons delivered by boat $ b $, route $r $, leg $l $ to district $j $ in quarter $t $}\\
    g_{b, r, l, t}&\ge 0 \quad\text{: tons transferred by boat $b $, route $r $, leg $l $ to destination warehouse\footnotemark}\\
    s_{w, t}&\ge 0 \quad\text{: inventory at warehouse $w $ at the end of quarter $t $}\\
    u_{j, t}&\ge 0 \quad\text{: unmet demand in district $j $ in quarter $ t $}
\end{align}
\footnotetext{Only when orig. $\neq $ dest.}
\subsubsection{Objective Function}
\begin{align}
    \min Z &= Z^{\text{raw}} + Z^{\text{shift}} + Z^{\text{ot}} + Z^{\text{store}} + Z^{\text{rent}} + Z^{\text{boat}} + Z^{\text{pen}}\\
    \shortintertext{where}\notag\\
    Z^{\text{raw}} &= \sum_{p\in \mathcal{P}}\sum_{t\in \mathcal{T}}c_p^{\text{raw}}x_{p, t}\\
    Z^{\text{shift}} &= \sum_{p\in \mathcal{P}}\sum_{t\in \mathcal{T}}c^{\text{shift}}n_{p, t}\\
    Z^{\text{ot}} &= \sum_{p\in \mathcal{P}}\sum_{t\in \mathcal{T}}\left(c^{\text{ot4}}\delta_{p, t}^4 + c^{\text{ot5}}\delta_{p, t}^5\right)\\
    Z^{\text{store}} &= \sum_{w\in \mathcal{W}}\sum_{t\in \mathcal{T}}c^{\text{store}}s_{w, t}\\
    Z^{\text{rent}}&= c^{\text{rent}}y\\
    Z^{\text{boat}} &= \sum_{b\in \mathcal{B}} c_b^{\text{year}}z_b + \sum_{b\in \mathcal{B}^H}c_b^{\text{half}}z_b^H\\
    Z^{\text{pen}} &= \sum_{j\in \mathcal{D}}\sum_{t\in \mathcal{T}}c^{\text{pen}}u_{j, t}
\end{align}
\newpage 
\begin{gather}
    \shortintertext{s.t.}
    \delta^5_{p, t} \le \delta^4_{p, t}\quad\forall\, p\in \mathcal{P}, t\in \mathcal{T}\\
    x_{p, t}\le \chi_p n_{p, t} \quad\forall\, p \in \mathcal{P}, t \in \mathcal{T}\\
    \sum_{b\in \mathcal{B}}\alpha_{b, r, t}\le 1\quad\forall\, r\in \mathcal{R}, t\in \mathcal{T}\\
    \sum_{r\in \mathcal{R}}\alpha_{b, r, t}\le 1\quad\forall\, b\in \mathcal{B}, t\in \mathcal{T}\\
    \alpha_{b,r,t} \leq z_b + z_b^H \quad\forall\, b \in \mathcal{B}^H, r \in \mathcal{R}, t \in \{2,3\}\\
    \alpha_{b,r,t} \leq z_b \quad \forall\, b \in \mathcal{B}^H, r \in \mathcal{R}, t \in \{1,4\}\\
    z_b + z_b^H \leq 1 \quad \forall\, b \in \mathcal{B}^H\\
    \alpha_{b,r,t} \leq z_b \quad \forall\, b \in \mathcal{B} \setminus \mathcal{B}^H, r \in \mathcal{R}, t \in \mathcal{T}\\
    \alpha_{b, r, t} \le y \quad\forall\, b\in \mathcal{B}, r\in \mathcal{R}^C, t\in \mathcal{T}\\
    \sum_{j\in \mathcal{V}(r, l)}f_{b, r, l, j, t} + g_{b, r, l, t}\le \bar{L}_b\frac{\text{days}_t}{D_r}\alpha_{b, r, t}\quad\forall\, b \in \mathcal{B}, r \in \mathcal{R}, l \in \mathcal{L}_r, t \in \mathcal{T}\\
    \sum_{\substack{b, r, l \,:\, j \in \mathcal{V}(r,l)}} f_{b,r,l,j,t} + u_{j,t} = d_{j,t}\quad\forall\, j\in \mathcal{D}, t\in \mathcal{T}\\
    s_{w, t} = s_{w, t - 1} + \text{prod}_{w, t} + \text{in}_{w, t} - \text{out}_{w, t}\quad\forall\, w \in \mathcal{W}, t \in \mathcal{T},\; s_{w,0} = 0\\
    \shortintertext{where\footnotemark}
    \text{prod}_{w, t} = 
    \begin{cases}
        x_{p,t} & \quad\text{if $w$ is co-located with plant $p$}\\
        0 & \quad\text{otherwise}
    \end{cases}\\
    \text{in}_{w, t} = \sum_{\substack{b, r, l \,:\, \text{dest}(r, l) = w\\\text{orig}(r, l) \neq w}} g_{b, r, l, t}\\
    \text{out}_{w, t} = \sum_{b, r, l \,:\, \text{orig}(r, l) = w}\left( \sum_{j\in \mathcal{V}(r, l)} f_{b, r, l, j, t} + g_{b, r, l, t} \mathbbm{1}\left[\text{orig}\neq\text{dest}\right] \right)\\
    s_{w, t}\le \bar{s}_w\quad\forall\, w\in \{N, S\}, t\in \mathcal{T}\\
    s_{C,t} \le \bar{s}_C \cdot y \quad \forall\, t \in \mathcal{T}\\
    \notag\\
    \shortintertext{Define handling at warehouse $w$ in quarter $t$ as:}\notag\\
    \text{handling}_{w,t} = 
    \underbrace{
        \sum_{\substack{b,r,l\;:\;\text{orig}(r,l)=w}}
        \left( \sum_{j \in \mathcal{V}(r,l)} f_{b,r,l,j,t} + g_{b,r,l,t}\mathbbm{1}[\text{orig}\neq\text{dest}] \right)
    }_{\text{loading at } w}
    +
    \underbrace{
        \sum_{\substack{b,r,l\;:\;\text{dest}(r,l)=w \\ \text{orig}(r,l) \neq w}}
        g_{b,r,l,t}
    }_{\text{discharging at } w}\\
    \text{handling}_{w,t} \le \bar{h}_w
    \quad \forall\, w \in \{N, S\}, t \in \mathcal{T}\\
    \text{handling}_{C,t} \le \bar{h}_C \cdot y \quad \forall\, t \in \mathcal{T}
\end{gather}
\footnotetext{The $g$ term is only present when $\text{orig}(r, l)\neq\text{dest}(r, l)$; on legs where $\text{orig}=\text{dest}$, $g_{b,r,l,t}=0$ by definition.\\Also I must apologize for the hectic alignment.}

\subsubsection{Justification of constraints}
\begin{enumerate}
    \setcounter{enumi}{47}
    \item Shift ordering: We can't run the $5^{\text{th}}$ shift without also running the $4^{\text{th}}$. 
    \item Production capacity: The total tons produced by plant $p $ in quarter $t $ cannot exceed the capacity per shift times the number of shifts being run 
    \item At most one boat per route per quarter
    \item At most one route per boat per quarter 
    \item Half-year eligible boats in Q2/Q3
    \item Half-year eligible boats in Q1/Q4: Anna and Emma can only be used in Q1/Q4 if they are rented for the full year. 
    \item Mutually exclusive rental options: For Anna and Emma, we can rent for the full year or half of the year, but not both
    \item Regular boats require full-year rental 
    \item Central warehouse routes require rental: No boat can be assigned to a route that visits the central warehouse unless the central warehouse is rented 
    \item Leg capacity: On each leg of a route, the total tons delivered to districts plus tons transferred to the destination warehouse cannot exceed that boat's capacity times the number of complete route cycles in the quarter. The boat carries at most $\bar{L}_b $ tons per leg per cycle, and completes $\text{days}_t/D_r $ cycles. When the boat isn't assigned $(\alpha = 0) $, the RHS is zero, forcing all flows to zero.
    \item Demand satisfaction: For each district in each quarter, the total deliveries from all boats plus the unmet demand must equal the demand. This is an equality, so any shortfall is captured by $u_{j, t} $, which feeds into the penalty cost
    \item Inventory balance: Ending inventory at each warehouse equals the previous quarter's inventory, plus production (if co-located with a plant), plus inbound transfers from other warehouses, minus outbound flows (deliveries to districts and transfers to other warehouses). Initial inventory is zero
    \item Helper binary variable 
    \item The sum of all transfers $g $ arriving at warehouse $w $ from other warehouses. Only counts legs where the destination is $w $ and the origin is different 
    \item The sum of everything loaded into boats at warehouse $w $, both deliveries to districts $(f) $ and transfers to other warehouses ($g $, gated by the indicator (characteristic) function $\mathbbm{1}[\text{orig}\neq\text{dest}] $ so that goods on round-trip legs back to the same warehouse aren't double-counted as outflow) 
    \item Plant storage capacity 
    \item Central warehouse capacity. Note the binary 
    \item Total loading plus total discharging at warehouse $w $. Note that it does not include moving product from production into the co-located warehouse (only boat-related movement)
    \item Handling capacities at $N\land S $
    \item Handling capacity at $C $
\end{enumerate}

\subsubsection{Should we rent the central warehouse?}
Here we don't actually need to set up a model, because the answer is trivial if you know where to look. 

The only quarter where demand exceeds max production is Q3, with a shortfall of 12,000 tons. Without the central warehouse, plant storage is 17,500 tons in total with $N + S $. This means we in good conscience can move on to the next bottleneck, which is transportation. 

The case for the central warehouse rests entirely on delivery capacity to districts 3 and 4. Without $C $, the available routes are 3, 4, 7 and 8. The only route that visits districts 3 and 4 is route 8. With the ``one-boat-per-route-per-quarter'' constraint, at most one boat can serve districts 3 and 4 in any given quarter. 

Best case: Anna (1,800 tons) on route 8 every quarter. Delivery rate $ = \frac{2\times 1800}{12} = 300 $ tons per day.

\subsubsection{Uncertainties}

There is a plethora of uncertainties that may arise. For instance, the demand figures presented in this model are explicitly described as ``estimated''. Fish feed demand depends on an array of factors, such as growth rates, disease outbreaks, market conditions, and customer ordering behavior. The seasonal pattern itself may shift from year to year. This is arguably the most impactful source of uncertainty, since the entire production and distribution plan is built around these numbers. 

In addition to that, there is the transportation and delivery uncertainties, where we assume that boats are always on schedule and never break down. In reality, boats do in fact get delayed and suffer mechanical errors, and they are especially prone to that in Norwegian conditions. Since delivery capacity is already a binding bottlenect, even modest disruptions could cause significant unmet demand. 

We also have uncertainties around the core production itself (equipment breakdown, unplanned maintenance, supply chain issues) and raw material prices. The deterministic model achieves zero penalty, but this is only guaranteed under the exact parameter values used. If demand turns out to be higher than estimated, or if a boat is delayed by weather, then districts would experience shortfalls.

There are several ways we can address the element of uncertainty in the model. If we instead model the demand (and potentially transportation capacity) as random variables with \textit{known distributions} instead of hard-coded values, we're able to use a two-stage stochastic program\footnotemark. We can also replace or bolster the penalty cost constraint by imposing explicit customer service requirements, like requiring that 95\% of demand needs to be met in each district in each quarter, across all scenarios. In addition to this, stocking up a safety storage at the warehouses is a trivial decisions at 50 NOK per ton. 
\footnotetext{Read more about this in Appendix A}

\subsection{Justification of constraints}
\begin{enumerate}
    \setcounter{enumi}{96}
    \item Production at site $i $ in period $t $ cannot exceed the number of active shifts times the production capacity per shift 
    \item Shift ordering: a later shift cannot be used unless the previous shift is also used. e.g., $5^{\text{th}}$ shift cannot come right after $3^{\text{rd}}$ shift
    \item Each site must run at least the minimum required number of shifts in every period 
    \item Each site can run no more thatn the maximum allowed number of shifts in every period 
    \item Inventory balance, first period: initial inventory plus production, minus deliveries to districts, minus outgoing internal transfers, plus incoming internal transfers, gives ending inventory in period 1
    \item Inventory balance, later periods: Same as above, except the starting inventory is the previous period's ending inventory
    \item Ending inventory at each site cannot exceed its storage capacity
    \item Handling capacity: outbound deliveries, outgoing internal transport, and incoming internal transport together cannot exceed the site's handling capacity in that period 
    \item Demand is either satisfied by deliveries from sites or recorded as lost sales. Equivalently, deliveries plus lost sales equal demand 
    \item Boat-route assignment: each boat can be assigned to at most one route per period, and each route can have at most one boat per period
    \item A boat can only be assigned to a route if that boat has been rented 
    \item Total deliveries from site $i $ to district $k $ are the sum of all route-leg deliveries that perform that movement
    \item Total internal transport from site $i $ to site $j $ is the sum of all route-leg internal shipments 
    \item Route-leg capacity: total district deliveries and internal shipments on a route leg cannot exceed the boat capacity times the number of route cycles possible in the period
    \item District delivery flow is only allowed if that route leg actually connects site $i $ to district $k $
    \item Internal transport flow is only allowed if that route leg actually connects site $i $ to site $j $
\end{enumerate}

\section{Cyclic Model / Static planning models}
\textbf{Assumptions}
\begin{itemize}
    \item Demand is constant and continuous: $D $ per year 
    \item The lot size (the decision variable) is $Q $
    \item A fixed cost (``order cost'' or ``setup cost'') $A $ is incurred every time a replenishment (order) takes place  
\end{itemize}
\textbf{Example}

$D = 1000 $ per year, $A = 200 $ per replenishment

If the lot size $Q = 100 $, then the number of replenishments per year is:
\begin{gather*}
    \frac{D}{Q} = \frac{1000}{100} = 10
\end{gather*}
Total order costs per year = $10\times 200 = 2000 $, and to minimize these fixed costs, maximize $Q $ to get as few replenishments as possible.

\textbf{But what about inventory holding costs?}

The inventory will follow a sawtooth pattern; at the maximum, the inventory will be $Q $, and on the minimum it will be $0 $. On average, the inventory level will be $Q / 2 $.

Inventory cost is assumed to be an annual ``inventory rate'' $r $, multiplied with the value of the product $v $.

Assume $r = 0.1 $ and $v = 500 $. Inventory holding cost per year for $Q = 100 $: 
\begin{gather*}
    v\times r \times \frac{Q}{2} = 500\times 0.1\times 100/2 = 2500 
\end{gather*}
Thus, \textbf{total relevant costs (TRC)}:
\begin{gather*}
    \text{TRC} = \underbrace{A\times \frac{D}{Q}}_{\text{total order costs}} + \underbrace{v\times r\times \frac{Q}{2}}_{\text{total inventory holding costs}}
\end{gather*}
In our numerical example, where $Q = 100 $:
\begin{gather*}
    \text{TRC} = 200\times 1000 / 100 + 500\times 0.1\times 100 / 2 = 4500
\end{gather*}
But which level of $Q $ will minimize TCR? \textbf{Take the derivative w.r.t. $Q$}:
\begin{gather*}
    \frac{\partial\text{TRC}}{\partial Q} = -\frac{A\times D}{Q^2} + \frac{v\times r}{2} = 0\\
    \therefore \boxed{Q = \sqrt{\frac{2\times A\times D}{v\times r}}}
\end{gather*}

The cost of serving (1) and (2) separately: 
\begin{gather*}
    C_{D, 1} + C_{1, D} + C_{D, 2} + C_{2, D} = 12 + 12 + 10 + 10 = 44

\begin{gather*}

\end{gather*}

\begin{gather*}

Then we rank them: $\{S_{1, 4}, S_{1, 2}, S_{2, 3}, S_{2, 4}, S_{1, 3}, S_{3, 4}\}$

\section{Stochastic Programming}
Optimal decisions should be based on data available at the time the decisions are made and cannot depend on future observations. The decision process is split into two stages:
\begin{itemize}
    \item Stage 1: ``here-and-now decisions''

    First-stage decision variables represent decisions that must be made before the uncertain paramaters are realized. Example: how much inventory to order, or whether or not we should rent a central warehouse.
    \item Stage 2: ``wait-and-see'' (recourse) decisions

    After uncertainty is revealed, recourse decisions are made to adapt. These are corrective or reactive actions taken to compensate for any mismatch between the first-stage decision and the realized scenario. Relevant example: overtime production or emergency shipments.
\end{itemize}
General formulation: 
\begin{gather*}
    \min_{z\in X}\{g(x) = f(x) + E_\xi[Q(x, \xi)]\}
\end{gather*}
where $Q(x, \xi)$ is the optimal value of the second-stage problem 
\begin{gather*}
    \min_y\{q(y, \xi)|T(\xi)x + W(\xi)y = h(\xi)\}
\end{gather*}
